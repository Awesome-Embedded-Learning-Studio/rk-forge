# Native production-code tests; included by src/rknpu/rk3588 with RKNN_BUILD_TESTS=ON.
foreach(name IN ITEMS execution_test output_test driver_test)
    add_executable(rknpu_${name} "native/${name}.cpp")
    target_link_libraries(rknpu_${name} PRIVATE rknpu_core)
    target_compile_options(rknpu_${name} PRIVATE -Wall -Wextra -Werror -ffp-contract=off -UNDEBUG)
endforeach()
add_test(NAME rknpu.driver COMMAND rknpu_driver_test)
set_tests_properties(rknpu.driver PROPERTIES ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
add_test(NAME rknpu.execution COMMAND rknpu_execution_test)
set_tests_properties(rknpu.execution PROPERTIES ENVIRONMENT "RKNN_MODEL_DIR=${RKNN_MODEL_DIR};ASAN_OPTIONS=detect_leaks=0")
if(RKNN_SYNC_EVIDENCE)
    add_test(NAME rknpu.output-snapshot COMMAND rknpu_output_test "${RKNN_SYNC_EVIDENCE}")
    set_tests_properties(rknpu.output-snapshot PROPERTIES ENVIRONMENT "RKNN_MODEL_DIR=${RKNN_MODEL_DIR};ASAN_OPTIONS=detect_leaks=0")
endif()
