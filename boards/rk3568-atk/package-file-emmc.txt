# package-file-emmc.txt — RKAF manifest for the RK3568 eMMC update.img.
#
# RK3568 eMMC update.img = RKFW download loader (MiniLoaderAll.bin, boot_merger from the
# public rkbin blobs — the `db` helper) + GPT (parameter-emmc-atk.txt) + partition images:
#   uboot   = u-boot.itb  (binman: mainline U-Boot + BL31/ATF, build-uboot.sh — NOT the
#                          fit-packed uboot.img aes uses; rkfw-pack stages it by basename)
#   boot    = boot.img    (kernel FIT: arm64 Image + rk3568-evb1-v10.dtb, pack-fit.sh)
#   rootfs  = rootfs.ext4 (buildroot busybox → stage-rootfs → pack-emmc.sh's mke2fs -d;
#                          staged as rootfs.img to match the manifest name)
# The eMMC @0x40 boot loader (binman idbloader.img) is NOT a partition here — RKDevTool
# writes idbloader raw at sector 0x40 via the loader's idb stage; this manifest only
# carries the GPT partitions + the download loader.

# NAME		Relative path
package-file	package-file
